import jakarta.persistence.*;

@Entity
@Table(name = "guard")
public class Guard {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(length = 100, nullable = false)
    private String name;

    @Column(length = 50, unique = true, nullable = false)
    private String username;

    @Column(length = 15, nullable = false)
    private String phone;

    @Column(length = 128, nullable = false)
    private String password;

    @Column(length = 30, nullable = false)
    private String shift = "Day";

    @Column(name = "is_active", nullable = false)
    private Boolean isActive = true;

    public Guard() { }

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    public String getUsername() { return username; }
    public void setUsername(String username) { this.username = username; }
    public String getPhone() { return phone; }
    public void setPhone(String phone) { this.phone = phone; }
    public String getPassword() { return password; }
    public void setPassword(String password) { this.password = password; }
    public String getShift() { return shift; }
    public void setShift(String shift) { this.shift = shift; }
    public Boolean getIsActive() { return isActive; }
    public void setIsActive(Boolean active) { isActive = active; }

    @Override
    public String toString() {
        return username;
    }
}

import com.example.society.entity.Guard;
import com.example.society.dto.GuardDto;
import com.example.society.exception.ResourceNotFoundException;
import com.example.society.repository.GuardRepository;
import com.example.society.util.PasswordUtil;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;

@Service
public class GuardService {
    private final GuardRepository guardRepository;

    public GuardService(GuardRepository guardRepository) {
        this.guardRepository = guardRepository;
    }

    public List<Guard> getAllGuards() {
        return guardRepository.findAll();
    }

    public Guard getGuardById(Long id) {
        return guardRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Guard not found with id: " + id));
    }

    public Guard createGuard(GuardDto dto) {
        Guard guard = new Guard();
        guard.setName(dto.getName());
        guard.setUsername(dto.getUsername());
        guard.setPhone(dto.getPhone());
        guard.setPassword(PasswordUtil.hashPassword(dto.getPassword()));
        guard.setShift(dto.getShift() != null ? dto.getShift() : "Day");
        guard.setIsActive(dto.getIsActive() != null ? dto.getIsActive() : true);
        return guardRepository.save(guard);
    }

    public Guard updateGuard(Long id, GuardDto dto) {
        Guard guard = getGuardById(id);
        guard.setName(dto.getName());
        guard.setUsername(dto.getUsername());
        guard.setPhone(dto.getPhone());
        if (dto.getPassword() != null && !dto.getPassword().trim().isEmpty()) {
            guard.setPassword(PasswordUtil.hashPassword(dto.getPassword()));
        }
        if (dto.getShift() != null) {
            guard.setShift(dto.getShift());
        }
        if (dto.getIsActive() != null) {
            guard.setIsActive(dto.getIsActive());
        }
        return guardRepository.save(guard);
    }

    public void deleteGuard(Long id) {
        Guard guard = getGuardById(id);
        guardRepository.delete(guard);
    }

    public Optional<Guard> login(String username, String password) {
        Optional<Guard> guardOpt = guardRepository.findByUsername(username);
        if (guardOpt.isPresent()) {
            Guard guard = guardOpt.get();
            if (PasswordUtil.checkPassword(password, guard.getPassword()) && Boolean.TRUE.equals(guard.getIsActive())) {
                return Optional.of(guard);
            }
        }
        return Optional.empty();
    }
}

import { Injectable } from '@angular/core';
import { Observable } from 'rxjs';

export interface Guard {
  id?: number;
  name: string;
  username: string;
  phone: string;
  password?: string;
  shift: string;
  isActive?: boolean;
}

@Injectable({
  providedIn: 'root'
})
export class GuardService {
  private endpoint = 'gaurds';

  constructor(private api: HttpClient) {}

  getAllGuards(): Observable<Guard[]> {
    return this.api.get<Guard[]>(this.endpoint);
  }

  getGuardById(id: number): Observable<Guard> {
    return this.api.get<Guard>(this.endpoint + '/' + id);
  }

  createGuard(guard: any): Observable<Guard> {
    return this.api.post<Guard>(this.endpoint, guard);
  }

  updateGuard(id: number, guard: any): Observable<Guard> {
    return this.api.put<Guard>(this.endpoint + '/' + id, guard);
  }

  deleteGuard(id: number): Observable<void> {
    return this.api.delete<void>(this.endpoint + '/' + id);
  }

  login(credentials: { username: string; password: string }): Observable<any> {
    return this.api.post<any>(this.endpoint + '/login', credentials);
  }
}

import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { GuardService } from '../../../services/guard.service';
import { Guard } from '../../../models/guard.model';
import Swal from 'sweetalert2';

@Component({
  selector: 'app-guards',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './guards.component.html',
  styleUrl: './guards.component.css'
})
export class GuardsComponent implements OnInit {
  guards: Guard[] = [];
  isEdit = false;
  success = '';
  error = '';
  currentGuard: Guard = { name: '', username: '', phone: '', password: '', shift: 'Day', isActive: true };

  constructor(private gs: GuardService) {}

  ngOnInit(): void {
    this.loadGuards();
  }

  loadGuards(): void {
    this.gs.getAllGuards().subscribe({
      next: (d) => this.guards = d,
      error: () => this.error = 'Failed to load guards.'
    });
  }

  saveGuard(): void {
    if (!this.currentGuard.name || !this.currentGuard.username) {
      this.error = 'Name and Username are required.';
      return;
    }
    this.gs.createGuard(this.currentGuard).subscribe({
      next: () => {
        this.success = 'Security Guard registered successfully!';
        this.resetForm();
        this.loadGuards();
      },
      error: () => this.error = 'Failed to create guard.'
    });
  }

  editGuard(g: Guard): void {
    this.isEdit = true;
    this.success = '';
    this.error = '';
    this.currentGuard = { ...g };
  }

  updateGuard(): void {
    if (!this.currentGuard.id) return;
    this.gs.updateGuard(this.currentGuard.id, this.currentGuard).subscribe({
      next: () => {
        this.success = 'Security Guard updated successfully!';
        this.resetForm();
        this.loadGuards();
      },
      error: () => this.error = 'Failed to update guard.'
    });
  }

  deleteGuard(id: number): void {
    Swal.fire({
      title: 'Are you sure?',
      text: 'Do you want to delete this guard?',
      icon: 'warning',
      showCancelButton: true,
      confirmButtonText: 'Yes, delete',
      confirmButtonColor: '#dc3545'
    }).then((res) => {
      if (res.isConfirmed) {
        this.gs.deleteGuard(id).subscribe({
          next: () => {
            this.success = 'Guard deleted successfully!';
            this.loadGuards();
          },
          error: () => this.error = 'Failed to delete guard.'
        });
      }
    });
  }

  resetForm(): void {
    this.isEdit = false;
    this.currentGuard = { name: '', username: '', phone: '', password: '', shift: 'Day', isActive: true };
    this.error = '';
  }
}

import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { Router, ActivatedRoute, RouterModule } from '@angular/router';
import { ResidentService } from '../../../services/resident.service';
import { GuardService } from '../../../services/guard.service';
import Swal from 'sweetalert2';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [CommonModule, FormsModule, RouterModule],
  templateUrl: './login.component.html',
  styleUrl: './login.component.css'
})
export class LoginComponent implements OnInit {
  selectedRole: 'RESIDENT' | 'GUARD' = 'RESIDENT';
  username = '';
  password = '';
  isLoading = false;

  constructor(
    private router: Router,
    private route: ActivatedRoute,
    private residentService: ResidentService,
    private guardService: GuardService
  ) {}

  ngOnInit(): void {
    const roleParam = this.route.snapshot.queryParams['role'];
    if (roleParam === 'guard') {
      this.selectedRole = 'GUARD';
    } else {
      this.selectedRole = 'RESIDENT';
    }
  }

  setRole(role: 'RESIDENT' | 'GUARD'): void {
    this.selectedRole = role;
    this.username = '';
    this.password = '';
  }

  onSubmit(): void {
    if (!this.username.trim() || !this.password.trim()) {
      Swal.fire({ icon: 'warning', title: 'Missing Info', text: 'Please fill both username/email and password.' });
      return;
    }
    this.isLoading = true;

    if (this.selectedRole === 'RESIDENT') {
      this.residentService.login({ username: this.username, password: this.password }).subscribe({
        next: (res) => {
          this.isLoading = false;
          localStorage.setItem('currentUser', JSON.stringify(res));
          Swal.fire({ icon: 'success', title: 'Welcome!', text: `Welcome back, ${res.name}`, timer: 1200, showConfirmButton: false })
            .then(() => this.router.navigate(['/resident/dashboard']));
        },
        error: (err) => {
          this.isLoading = false;
          Swal.fire({ icon: 'error', title: 'Login Failed', text: err.error || 'Invalid resident email or password.' });
        }
      });
    } else {
      this.guardService.login({ username: this.username, password: this.password }).subscribe({
        next: (res) => {
          this.isLoading = false;
          localStorage.setItem('currentUser', JSON.stringify(res));
          Swal.fire({ icon: 'success', title: 'Duty Active!', text: `Welcome Guard ${res.name}`, timer: 1200, showConfirmButton: false })
            .then(() => this.router.navigate(['/guard/dashboard']));
        },
        error: (err) => {
          this.isLoading = false;
          Swal.fire({ icon: 'error', title: 'Login Failed', text: err.error || 'Invalid guard username or password.' });
        }
      });
    }
  }
}
